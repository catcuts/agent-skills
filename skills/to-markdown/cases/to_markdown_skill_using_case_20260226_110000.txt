  一、问题总结

  1.1 下载机制问题

  问题 1：反爬限制导致 urllib 失败
  - 现象：urllib 下载的 HTML 不包含实际内容（JavaScript 未渲染）
  - 原因：x.com 等现代网站依赖 JavaScript 动态加载内容
  - 影响：模式1（URL 下载）在大多数情况下失败

  问题 2：自动降级机制不完善
  - 现象：download_html.py 尝试调用 agent-browser 失败
  - 原因：
    - Windows 上 agent-browser 命令不在 PATH 中
    - npx 调用方式需要特殊处理
    - 错误处理不够健壮
  - 影响：降级到 Playwright 方案失败

  问题 3：Playwright 脚本 bug
  - 现象：download_html_playwright.js 报错 stderr is not defined
  - 原因：代码中使用了未定义的 stderr 变量
  - 影响：第三降级方案也失败

  1.2 Cookie 认证问题

  问题 4：环境变量 Cookie 未正确应用
  - 现象：设置了 COOKIE_x_com 但页面仍显示未登录状态
  - 原因：
    - --headers 参数 JSON 格式解析失败（Cookie 含特殊字符）
    - PowerShell 脚本 set-cookies-from-string.ps1 参数传递问题
    - agent-browser 与 urllib 的 Cookie 机制不同
  - 影响：无法加载需要登录的内容（如评论）

  问题 5：Cookie 设置方法不明确
  - 现象：文档提到 COOKIE_<域名> 环境变量，但实际使用时需要手动调用 cookies set
  命令
  - 原因：文档与实际实现不匹配
  - 影响：用户困惑，需要手动调试

  1.3 内容提取问题

  问题 6：评论加载需要特殊处理
  - 现象：主推文可获取，但评论需要滚动或点击才能加载
  - 原因：懒加载机制
  - 影响：获取的内容不完整

  问题 7：页面状态不稳定
  - 现象：滚动后提取的内容顺序错乱或内容不对
  - 原因：
    - DOM 元素顺序变化
    - 页面异步加载导致内容不一致
  - 影响：需要多次尝试才能正确提取

  1.4 用户体验问题

  问题 8：错误信息不够友好
  - 现象：技术错误直接暴露给用户
  - 原因：缺少用户友好的错误处理
  - 影响：用户不知道如何解决问题

  问题 9：进度反馈不足
  - 现象：长时间等待无反馈
  - 原因：缺少进度指示
  - 影响：用户体验差

  #!/usr/bin/env python3
  """
  自动为 agent-browser 设置 cookies
  从环境变量 COOKIE_<域名> 读取并设置
  """
  import os
  import sys
  import subprocess
  import json
  from urllib.parse import urlparse

  def get_domain_from_url(url):
      """从 URL 提取域名"""
      parsed = urlparse(url)
      return parsed.netloc

  def set_cookies_for_domain(domain):
      """为指定域名设置 cookies"""
      # 环境变量名格式：x.com -> COOKIE_x_com
      env_name = f"COOKIE_{domain.replace('.', '_')}"
      cookie_string = os.environ.get(env_name, '')

      if not cookie_string:
          print(f"No cookies found for {domain} (env var: {env_name})",
  file=sys.stderr)
          return False

      # 解析 cookie 字符串
      cookies = []
      for part in cookie_string.split(';'):
          part = part.strip()
          if not part:
              continue

          if '=' in part:
              name, value = part.split('=', 1)
              cookies.append((name.strip(), value.strip()))

      # 使用 agent-browser 设置每个 cookie
      success_count = 0
      for name, value in cookies:
          try:
              # 跳过过长的 cookie 值（可能导致错误）
              if len(value) > 1000:
                  print(f"Skipping long cookie: {name}", file=sys.stderr)
                  continue

              result = subprocess.run(
                  ['npx', '--yes', 'agent-browser', 'cookies', 'set', name,
  value, '--url', f'https://{domain}'],
                  capture_output=True,
                  timeout=10,
                  text=True
              )

              if result.returncode == 0:
                  success_count += 1
                  print(f"✓ Set cookie: {name}", file=sys.stderr)
              else:
                  print(f"✗ Failed to set: {name}", file=sys.stderr)

          except Exception as e:
              print(f"✗ Error setting {name}: {e}", file=sys.stderr)

      print(f"Set {success_count}/{len(cookies)} cookies for {domain}",
  file=sys.stderr)
      return success_count > 0

  if __name__ == '__main__':
      if len(sys.argv) < 2:
          print("Usage: python setup_cookies.py <url>", file=sys.stderr)
          sys.exit(1)

      url = sys.argv[1]
      domain = get_domain_from_url(url)

      if set_cookies_for_domain(domain):
          sys.exit(0)
      else:
          sys.exit(1)

      (async () => {
        // 1. 验证主推文
        const firstArticle = document.querySelector('article');
        const author = 
  firstArticle?.querySelector('[data-testid="User-Name"]')?.textContent;

  """
  统一的下载接口
  自动选择最佳下载方法
  """
  import os
  import sys
  from urllib.parse import urlparse

  def download_html_unified(url, wait_for_js=True, scroll=False):
      """
      统一的 HTML 下载接口
      
      Args:
          url: 目标 URL
          wait_for_js: 是否等待 JavaScript 渲染
          scroll: 是否滚动加载更多内容
      
      Returns:
          (html_content, method_used)

      domain = urlparse(url).netloc

      # 策略1：静态网站（不需要 JS）
      if not wait_for_js:

      # 策略3：动态网站（需要 JS）
      if is_dynamic_site(url):
          if check_agent_browser_available():
              return download_with_agent_browser(url, scroll), 'agent-browser'
          elif check_playwright_available():
              return download_with_playwright(url, scroll), 'playwright'

      # 策略4：回退到 urllib
      return download_with_urllib(url), 'urllib'

  def has_cookies_for_domain(domain):
      """检查是否为该域名配置了 cookies"""

      return bool(os.environ.get(env_name))

  def is_dynamic_site(url):
      """判断是否是需要 JS 的动态网站"""
      dynamic_patterns = ['x.com', 'twitter.com', 'weibo.com', 'zhihu.com']

  #!/usr/bin/env python3
  """
  友好的错误处理和用户提示
  """

  class DownloadError(Exception):
      def __init__(self, message, suggestion=None):
          self.message = message
          self.suggestion = suggestion
          super().__init__(self.format_message())

      def format_message(self):
          msg = f"❌ {self.message}"
          if self.suggestion:
              msg += f"\n\n💡 建议：{self.suggestion}"
          return msg

  class CookieError(DownloadError):
      pass

  class NetworkError(DownloadError):
      pass

  def handle_error(error):
      """处理错误并提供友好提示"""
      if isinstance(error, CookieError):
          print("\n" + "="*60)
          print("🍪 Cookie 认证失败")
          print("="*60)
          print(error.message)
          print("\n解决方法：")
          print("1. 确保已设置环境变量 COOKIE_x_com")
          print("2. 运行：python setup_cookies.py https://x.com/...")
          print("="*60 + "\n")

      elif isinstance(error, NetworkError):
          print("\n" + "="*60)
          print("🌐 网络连接失败")
          print("="*60)
          print(error.message)
          print("\n可能的解决方案：")
          print("1. 检查网络连接")
          print("2. 尝试使用代理（设置 PLAYWRIGHT_PROXY）")
          print("3. 使用剪贴板模式（直接复制页面内容）")
          print("="*60 + "\n")

      else:
          print(f"\n❌ 错误：{error}\n")

  #!/usr/bin/env python3
  """
  进度显示工具
  """
  import sys
  import time

  class ProgressIndicator:
      def __init__(self, total_steps):
          self.total_steps = total_steps
          self.current_step = 0

      def advance(self, message):
          """前进到下一步"""
          self.current_step += 1
          percent = int(self.current_step / self.total_steps * 100)

          # 动画效果
          animations = ['⠋', '⠙', '⠹', '⠸', '⠼', '⠴', '⠦', '⠧', '⠇', '⠏']
          anim = animations[self.current_step % len(animations)]

          sys.stdout.write(f"\r{anim} [{self.current_step}/{self.total_steps}] 
  {message} ({percent}%)")
          sys.stdout.flush()

      def complete(self, message="完成！"):
          """完成"""
          sys.stdout.write(f"\r✓ {message}\n")
          sys.stdout.flush()

  # 使用示例
  def download_with_progress(url):
      progress = ProgressIndicator(5)

      progress.advance("正在下载页面...")
      html = download_html(url)

      progress.advance("正在解析 HTML...")
      soup = BeautifulSoup(html, 'html.parser')

      progress.advance("正在提取内容...")
      content = extract_content(soup)

      progress.advance("正在转换为 Markdown...")
      markdown = convert_to_markdown(content)

      progress.complete("转换完成！")

      return markdown
